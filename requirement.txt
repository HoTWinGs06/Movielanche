# ==============================================================================
# Movielanche - Project Requirements Specification
# Stack: Node.js + Express + PostgreSQL
# ==============================================================================

# ------------------------------------------------------------------------------
# 1. System & Runtime Prerequisites
# ------------------------------------------------------------------------------
# Node.js:     >= 18.0.0 (compatible with v20.x, v22.x, v24.x)
# Package Mgr: npm >= 9.0.0
# Database:    PostgreSQL >= 15.0 (Local Docker instance or Neon Serverless PostgreSQL)

# ------------------------------------------------------------------------------
# 2. Application Dependencies (Node.js / backend/package.json)
# ------------------------------------------------------------------------------
bcryptjs==3.0.3
cors==2.8.6
dotenv==18.0.6
express==5.2.1
google-auth-library==11.2.0
helmet==8.3.0
jsonwebtoken==9.0.3
pg==8.23.1

# ------------------------------------------------------------------------------
# 3. Environment Variables (backend/.env / .env.example)
# ------------------------------------------------------------------------------
# --- Database Configuration (PostgreSQL / Neon) ---
# DATABASE_URL=postgresql://<user>:<password>@<host>/movies?sslmode=require
# DB_USERNAME=<db_user>
# PASSWORD=<db_password>
# DB_NAME=movies
# HOST=<db_host>
# PORT=5432

# --- Authentication & Google OAuth ---
# JWT_SECRET=<secure_random_secret_minimum_32_characters>
# GOOGLE_CLIENT_ID=YOUR_GOOGLE_CLIENT_ID.apps.googleusercontent.com

# --- Optional Server & Runtime Settings ---
# PORT=5000
# CINEMA_TZ=Asia/Kolkata
# RATE_LIMIT_MAX=1
# NODE_ENV=development
# TRUST_PROXY=1
# CORS_ORIGIN=*
# DB_SSL_REJECT_UNAUTHORIZED=true

# ------------------------------------------------------------------------------
# 4. Quick Setup & Execution Guide
# ------------------------------------------------------------------------------
# Install dependencies:
#   cd backend
#   npm install
#   (Note on Windows PowerShell: use `npm.cmd install` if script execution is restricted)
#
# Configure Environment:
#   cp .env.example backend/.env    # Fill in database credentials, JWT_SECRET, and GOOGLE_CLIENT_ID
#
# Run server:
#   cd backend
#   npm start         # Runs on http://localhost:5000 (or specified PORT)
#
# Run tests:
#   cd backend
#   npm test          # Verifies frontend-backend contract
#   npm run test:api  # Runs API integration regression tests
